/* Lessons in a course whose target is written in another script than the learner's: the reading of a word is written under it, as on the cards. Readings come only from the course's own cards and lesson example lines (whole lines, then words or characters that always carry one reading there); a word with no reading in the course stays bare. Nothing is derived by rule. */
const RD_CLS={zh:'\\u3400-\\u9fff\\u{20000}-\\u{2fa1f}',ja:'\\u3040-\\u30ff\\u3400-\\u9fff',th:'\\u0e00-\\u0e7f',ar:'\\u0600-\\u060b\\u060d-\\u061e\\u0620-\\u06d3\\u06d5-\\u06ff\\u0750-\\u077f\\ufb50-\\ufdff\\ufe70-\\ufeff',hi:'\\u0900-\\u0963\\u0966-\\u097f',bn:'\\u0980-\\u09ff',pa:'\\u0a00-\\u0a7f',te:'\\u0c00-\\u0c7f',kn:'\\u0c80-\\u0cff',am:'\\u1200-\\u1360\\u1369-\\u139f',ru:'\\u0400-\\u04ff\\u0301'}; RD_CLS.yue=RD_CLS.nan=RD_CLS.zh; RD_CLS.ur=RD_CLS.ar;
const RD_HAR=/[\u064b-\u0652\u0670]/g, RD_TIGHT={zh:1,yue:1,nan:1,th:1,ja:1}, RD_HAN={zh:1,yue:1,nan:1};
function rdOn(){ const c=COURSE; return !!(c&&RD_CLS[c.target]&&!RD_CLS[c.native]&&c.native!=='ko'&&c.DECKS); }
function rdKey(s){ let k=String(s).normalize('NFC').toLowerCase().replace(/[\s ]+/g,' ').replace(/[.,!?;:…。！？、，；：·«»"“”‘’'()\[\]،؟۔।॥።፣፤፥፦፧—–~\/-]/g,' ').replace(/ +/g,' ').trim(); const t=COURSE.target; if(t==='ar'||t==='ur') k=k.replace(/\u0640/g,''); if(t==='ru') k=k.replace(/́/g,''); return RD_TIGHT[t]&&t!=='ja'?k.replace(/ /g,''):k; }
function rdEdge(x){ return String(x).replace(/^[\s.,!?;:…"“”()\/]+|[\s.,!?;:…"“”()\/]+$/g,''); }
function rdBuild(){ const c=COURSE; if(c._rd) return c._rd; const W=new Map(), C=new Map(), B=new Map(), bare=c.target==='ar'||c.target==='ur', tight=!!RD_TIGHT[c.target], han=!!RD_HAN[c.target], lat=/[a-zA-ZÀ-ɏḀ-ỿ]/;
  const cand=(k,r)=>{ if(!k||!r) return; let m=C.get(k); if(!m){ m=new Map(); C.set(k,m); } m.set(r,(m.get(r)||0)+1); };
  const add=(t,r)=>{ if(typeof t!=='string'||typeof r!=='string'||!lat.test(r)) return; const k=rdKey(t), rr=rdEdge(r.replace(/\s+/g,' ')); if(!k||!rr) return; const kw=k.replace(/ /g,tight?'':' '); if(!W.has(kw)) W.set(kw,rr); if(bare){ const kb=kw.replace(RD_HAR,''); if(kb!==kw){ let m=B.get(kb); if(!m){ m=new Set(); B.set(kb,m); } m.add(rr.toLowerCase()); B.set(kb+'\u0000',rr); } }
    if(han){ const ch=Array.from(kw), sy=rr.toLowerCase().split(/[\s\/-]+/).map(rdEdge).filter(Boolean); if(ch.length===sy.length) ch.forEach((x,i)=>cand(x,sy[i])); }
    else { const tt=k.split(' '), rt=rr.split(/[\s\/]+/).map(rdEdge).filter(Boolean); if(tt.length===rt.length&&tt.length>1) tt.forEach((x,i)=>cand(x,rt[i].toLowerCase())); } };
  Object.keys(c.DECKS||{}).forEach(d=>(c.DECKS[d]||[]).forEach(x=>{ if(x) add(x[0],x[1]); }));
  (c.LESSONS||[]).forEach(l=>String(l.body||'').replace(/\{\{ex:([^|}]*)\|([^|}]*)\|/g,(m,t,r)=>{ add(t,r); return m; }));
  const D=new Map(); C.forEach((m,k)=>{ if(m.size===1&&!W.has(k)) D.set(k,m.keys().next().value); }); B.forEach((m,k)=>{ if(m instanceof Set&&m.size===1&&!W.has(k)&&!D.has(k)) D.set(k,B.get(k+'\u0000')); });
  const cls=RD_CLS[c.target]; return c._rd={W,D,tight,han,rx:new RegExp('['+cls+']+'+(tight&&c.target!=='ja'?'':'(?:[ \\u00a0]['+cls+']+)*'),'gu')}; }
function rdTight(s,R){ const get=k=>R.W.get(k)||R.D.get(k)||null, k=rdKey(s).replace(/ /g,''), ch=Array.from(s), n=ch.length, whole=R.W.get(k)||null; if(whole&&n<=6) return [[s,whole]]; if(n>40) return [[s,whole]];
  const best=new Array(n+1).fill(null); best[0]={c:0,p:-1,r:null}; for(let i=0;i<n;i++){ if(!best[i]) continue; for(let j=i+1;j<=Math.min(n,i+12);j++){ const r=get(rdKey(ch.slice(i,j).join('')).replace(/ /g,'')); let cost=null; if(r) cost=1; else if(R.han&&j===i+1) cost=4; if(cost==null) continue; const c=best[i].c+cost; if(!best[j]||c<best[j].c) best[j]={c,p:i,r}; } }
  if(!best[n]) return [[s,whole]]; const out=[]; let j=n; while(j>0){ const b=best[j]; out.unshift([ch.slice(b.p,j).join(''),b.r]); j=b.p; } if(whole&&out.some(x=>!x[1])) return [[s,whole]]; return out; }
function rdSeg(run){ const R=rdBuild(); if(R.tight&&!/[  ]/.test(run)) return rdTight(run,R); const get=k=>R.W.get(k)||R.D.get(k)||null, parts=run.split(/([  ]+)/), words=parts.filter((x,i)=>i%2===0); const whole=R.W.get(rdKey(run).replace(/ /g,R.tight?'':' '))||null; if(words.length===1) return [[run,whole||get(rdKey(run))]];
  let per=null; if(whole){ const rt=whole.split(/[\s\/]+/).map(rdEdge).filter(Boolean); if(rt.length===words.length) per=rt; } const out=[]; parts.forEach((x,i)=>{ if(i%2){ out.push([x,null]); return; } if(per){ out.push([x,per[i/2]]); return; } const r=get(rdKey(x)); if(r||!R.tight) out.push([x,r]); else rdTight(x,R).forEach(y=>out.push(y)); }); return out; }
function rdAfter(nd,n){ let s='', x=nd; const root=document.body; while(s.length<n&&x&&x!==root){ let y=x.nextSibling; while(!y&&x.parentNode&&x.parentNode!==root){ x=x.parentNode; y=x.nextSibling; } if(!y) break; x=y; s+=x.textContent||''; } return s.slice(0,n); }
function rdNorm(s){ return String(s).toLowerCase().normalize('NFC').replace(/[\s.,!?;:…"“”()\[\]\/=-]+/g,''); }
function rdAnnotate(root){ const rtl=COURSE.target==='ar'||COURSE.target==='ur', R=rdBuild(), SK='.ex .k,.ex .r,.ex .sp,.sgl,ruby,rt,svg,script,style,input,textarea,.rom,[data-norom]'; const tw=document.createTreeWalker(root,NodeFilter.SHOW_TEXT), nodes=[]; let n; R.rx.lastIndex=0; while((n=tw.nextNode())){ R.rx.lastIndex=0; if(R.rx.test(n.nodeValue)&&n.parentElement&&!n.parentElement.closest(SK)) nodes.push(n); }
  nodes.forEach(nd=>{ const t=nd.nodeValue, frag=document.createDocumentFragment(); let last=0, m, any=false; R.rx.lastIndex=0; const hits=[]; while((m=R.rx.exec(t))) hits.push([m.index,m[0]]);
    hits.forEach(h=>{ const segs=rdSeg(h[1]); if(!segs.some(s=>s[1])) return; const full=segs.map(s=>s[1]||'').join(''), tail=rdNorm(t.slice(h[0]+h[1].length,h[0]+h[1].length+full.length+8)+(t.length-(h[0]+h[1].length)<full.length+8?rdAfter(nd,full.length+8):'')); if(full.length>1&&tail.indexOf(rdNorm(full))===0){ rdAnnotate.sk=(rdAnnotate.sk||0)+1; return; } if(full.length<=1&&rdNorm(tail).charAt(0)===rdNorm(full)&&!/[a-z]/.test(tail.charAt(1)||'')) return;
      frag.append(t.slice(last,h[0])); let host=frag; if(rtl&&segs.length>1){ host=document.createElement('bdi'); host.setAttribute('dir','rtl'); host.className='rbr'; frag.append(host); } segs.forEach(s=>{ if(!s[1]){ host.append(s[0]); return; } const rb=document.createElement('ruby'); rb.className='rb'; rb.append(s[0]); const rt=document.createElement('rt'); rt.setAttribute('dir','ltr'); rt.textContent=s[1]; rb.append(rt); host.append(rb); }); last=h[0]+h[1].length; any=true; });
    if(any){ frag.append(t.slice(last)); nd.replaceWith(frag); } }); }
/* a quick-check question is left bare when a reading would give the answer away: the reading of something on screen is itself among the answers, or the question names the reading of the right answer */
function rdSpoils(q){ try{ const R=rdBuild(), reads=s=>{ const o=[]; R.rx.lastIndex=0; let m; const str=String(s); while((m=R.rx.exec(str))) rdSeg(m[0]).forEach(x=>{ if(x[1]) x[1].toLowerCase().split(/[\s\/-]+/).map(rdEdge).forEach(w=>{ if(w.length>1) o.push(w); }); }); return o; }, latin=s=>' '+String(s).toLowerCase().replace(/<[^>]+>/g,' ').replace(/[^a-zÀ-ɏḀ-ỿ0-9]+/g,' ')+' ', has=(s,w)=>latin(s).indexOf(' '+w+' ')>=0;
  const opts=q[1].map(String), all=reads(q[0]).concat(...opts.map(reads)); if(opts.some(o=>all.some(w=>has(o,w)))) return true; return reads(opts[q[2]]).some(w=>has(q[0],w)); }catch(e){ return true; } }
